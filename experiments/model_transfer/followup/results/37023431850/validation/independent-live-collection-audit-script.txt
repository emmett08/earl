def same_scalar(a,b):
    return (type(a) is type(b) or (type(a) in (int,float) and type(b) in (int,float))) and a==b

def independent_reference(case, snap):
    states={}
    for req in case['rule']['requirements']:
        available=[f for f in snap['facts'] if f['key']==req['fact_key'] and f['status']=='active'
                   and all(k in f['scope'] and k in snap['scope'] and same_scalar(f['scope'][k],snap['scope'][k]) for k in req['scope_keys'])
                   and 0 <= snap['now_minute']-f['observed_at_minute'] <= req['max_age_minutes']]
        if not available: states[req['id']]=None; continue
        latest=max(f['observed_at_minute'] for f in available)
        chosen=[f for f in available if f['observed_at_minute']==latest]
        assert len(chosen)==1, 'newest eligible fact must be unique'
        value=chosen[0]['value'];bound=req['expected_value']
        if req['operator']=='eq': val=same_scalar(value,bound)
        else:
            assert type(value) in (int,float) and type(bound) in (int,float)
            val=value<=bound if req['operator']=='lte' else value>=bound
        states[req['id']]=val
    def both(xs): return False if any(v is False for v in xs) else None if any(v is None for v in xs) else True
    if case['rule']['type']=='all_of': routes={'all':both(list(states.values()))}; verdict=routes['all']
    else:
        routes={r['id']:both([states[k] for k in r['requirement_ids']]) for r in case['rule']['alternatives']}
        verdict=True if any(v is True for v in routes.values()) else None if any(v is None for v in routes.values()) else False
    if not all(k in snap['scope'] and same_scalar(snap['scope'][k],case['specification']['scope'][k]) for k in case['specification']['identity_scope_keys']):verdict=None
    return {'decision':'undetermined' if verdict is None else 'ready' if verdict else 'not_ready','requirement_states':states,'route_states':routes}

def normalised_prompt(messages):
    # The declared reasoner treatment adds only a computed decision to this packet.
    result=[]
    for message in messages:
        item=dict(message);prefix='Current task facts and any supplied computed conclusion:\n'
        if isinstance(item.get('content'),str) and item['content'].startswith(prefix):
            packet=json.loads(item['content'][len(prefix):]);packet.pop('decision',None)
            item['content']=prefix+json.dumps(packet,sort_keys=True,separators=(',',':'))
        result.append(item)
    return result

def dict_keys(value):
    if isinstance(value,dict):
        for k,v in value.items():yield k;yield from dict_keys(v)
    elif isinstance(value,list):
        for child in value:yield from dict_keys(child)
